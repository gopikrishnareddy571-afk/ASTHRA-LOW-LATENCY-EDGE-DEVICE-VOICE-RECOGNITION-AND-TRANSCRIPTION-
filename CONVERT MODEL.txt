import tensorflow as tf
import numpy as np
import os

MODEL_PATH = "model/asthra_model.keras"
TFLITE_PATH = "model/asthra_model_int8.tflite"

print("=" * 60)
print("       ASTHRA STATIC INT8 CONVERSION")
print("=" * 60)

# ------------------------------------------------------------
# LOAD MODEL
# ------------------------------------------------------------

print("\nLoading model...")

model = tf.keras.models.load_model(MODEL_PATH)

print("Input :", model.input_shape)
print("Output:", model.output_shape)


# ------------------------------------------------------------
# LOAD REPRESENTATIVE DATA
# ------------------------------------------------------------

X = np.load("model/X.npy").astype(np.float32)

if X.ndim == 3:
    X = X[..., np.newaxis]

print("Representative data:", X.shape)


def representative_dataset():
    for i in range(min(100, len(X))):
        yield [X[i:i + 1]]


# ------------------------------------------------------------
# FIXED INPUT SIGNATURE
# ------------------------------------------------------------

@tf.function(
    input_signature=[
        tf.TensorSpec(
            shape=[1, 13, 151, 1],
            dtype=tf.float32
        )
    ]
)
def serving_function(x):
    return model(x, training=False)


concrete_function = serving_function.get_concrete_function()


# ------------------------------------------------------------
# CONVERTER
# ------------------------------------------------------------

print("\nCreating converter...")

converter = tf.lite.TFLiteConverter.from_concrete_functions(
    [concrete_function]
)

converter.optimizations = [
    tf.lite.Optimize.DEFAULT
]

converter.representative_dataset = representative_dataset

converter.target_spec.supported_ops = [
    tf.lite.OpsSet.TFLITE_BUILTINS_INT8
]

converter.inference_input_type = tf.int8
converter.inference_output_type = tf.int8


# ------------------------------------------------------------
# CONVERT
# ------------------------------------------------------------

print("\nConverting...")

tflite_model = converter.convert()


# ------------------------------------------------------------
# SAVE
# ------------------------------------------------------------

os.makedirs("model", exist_ok=True)

with open(TFLITE_PATH, "wb") as f:
    f.write(tflite_model)

print("\nConversion complete!")
print(
    "Size:",
    f"{len(tflite_model) / 1024:.2f} KB"
)

print("Saved:", TFLITE_PATH)


# ------------------------------------------------------------
# CHECK OPERATORS
# ------------------------------------------------------------

print("\nChecking operators...")

interpreter = tf.lite.Interpreter(
    model_content=tflite_model
)

interpreter.allocate_tensors()

for op in interpreter._get_ops_details():
    print(" -", op["op_name"])


# ------------------------------------------------------------
# INPUT / OUTPUT
# ------------------------------------------------------------

inp = interpreter.get_input_details()[0]
out = interpreter.get_output_details()[0]

print("\nINPUT")
print("Shape:", inp["shape"])
print("Type :", inp["dtype"])

print("\nOUTPUT")
print("Shape:", out["shape"])
print("Type :", out["dtype"])

print("\n" + "=" * 60)
print("CHECK COMPLETE")
print("=" * 60)
